    class Newborn:
    def __init__(self, height, weight):
        self.height = height  # 单位: 厘米
        self.weight = weight  # 单位: 千克
        self.is_sleeping = False
        self.is_hungry = True
        self.needs_diaper_change = False
        self.needs_burping = False

    def sleep(self):
        self.is_sleeping = True
        self.is_hungry = False
        print("新生儿正在睡觉。")

    def wake_up(self):
        self.is_sleeping = False
        self.is_hungry = True
        print("新生儿醒了，可能需要喂食。")

    def feed(self):
        self.is_hungry = False
        self.needs_diaper_change = True
        self.needs_burping = True
        print("新生儿正在吃奶。")

    def burp(self):
        self.needs_burping = False
        print("新生儿拍嗝结束。")

    def change_diaper(self):
        self.needs_diaper_change = False
        print("尿布已更换。")

    def bathe(self):
        print("新生儿洗澡。")

    def routine(self):
        import time
        while True:
            self.wake_up()
            self.feed()
            time.sleep(10)  # 假设喂奶和拍嗝用时 10 秒
            self.burp()
            if self.needs_diaper_change:
                self.change_diaper()
            self.sleep()
            time.sleep(10800)  # 模拟新生儿每3小时醒一次


# 示例使用
baby = Newborn(51.2, 3.5)  # 创建一个新生儿实例
baby.routine()  # 开始新生儿的日常生活模拟
